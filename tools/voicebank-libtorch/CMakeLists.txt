cmake_minimum_required(VERSION 3.20)
project(voicebank_generator LANGUAGES CXX)
find_package(Torch 2.7 REQUIRED)
add_library(voicebank_generator SHARED generator.cpp)
target_compile_features(voicebank_generator PRIVATE cxx_std_17)
target_link_libraries(voicebank_generator PRIVATE "${TORCH_LIBRARIES}")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
if(MSVC)
  target_compile_options(voicebank_generator PRIVATE /bigobj)
endif()
